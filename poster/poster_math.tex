\block{28}{356}{381}{158}{%
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{@{}p{87mm}>{\raggedright\arraybackslash}p{281mm}@{}}
$\mathcal L$ & Four lockers, indexed by $l$.\\
$\mathcal S$ & $\{\mathrm{TD},\mathrm{STD}\}$: Two-Day and Standard.\\

\panel{24}{740}{3}{Integer linear optimization model}
\fill[pale] (24,765) rectangle (413,982);
\block{30}{771}{377}{204}{%
\subhead{Maximize total accepted requests}
\begin{equation}\max_{y}\quad Z(y)=\sum_{l\in\mathcal L}\sum_{s\in\mathcal S}\sum_{t\in\mathcal T}y_{lst}.
\tag{1}\end{equation}
\subhead{Demand bounds and indivisible packages}
\begin{equation}0\le y_{lst}\le d_{lst},\quad y_{lst}\in\mathbb Z,
\qquad\forall l,s,t.\tag{2}\end{equation}
\subhead{Expected occupancy cannot exceed capacity}
\begin{equation}
\sum_{s\in\mathcal S}\sum_{v=1}^{t}q_{s,t-v}\,y_{lsv}\le C_l,
\qquad \forall l,t.\tag{3}
\end{equation}
\note{Every earlier arrival cohort contributes its surviving share to day $t$. Coefficients are fixed; the objective and constraints are linear.}}
\block{28}{994}{381}{108}{%
\subhead{Problem size and structure}
\[|\mathcal L||\mathcal S||\mathcal T|=4\cdot2\cdot7=56
\quad\text{integer variables},\]
\[|\mathcal L||\mathcal T|=28\quad\text{capacity inequalities}.
\]
\note{Plus 56 demand upper bounds and nonnegativity. No capacity transfers between lockers; cross-day coupling occurs within each locker.}}

% RIGHT: convolution interpretation and exact numerical illustration.
\panel{428}{330}{4}{Occupancy as a convolution}
\block{432}{356}{381}{204}{%
\subhead{Expected reservations by option}
\begin{equation}
x_{lst}=\sum_{v=1}^{t}q_{s,t-v}\,y_{lsv},\qquad
O_{lt}=\sum_{s\in\mathcal S}x_{lst}.\tag{4}
\end{equation}
\note{$x_{lst}$ and $O_{lt}$ are derived expected slot counts; they may be fractional. Only acceptances $y_{lst}$ are integer.}
\subhead{Base-case capacity constraint, expanded}
\begin{align*}
O_{lt}={}&y_{l,\mathrm{TD},t}+y_{l,\mathrm{STD},t}\\
 &+0.25y_{l,\mathrm{TD},t-1}+0.70y_{l,\mathrm{STD},t-1}\\
 &+0.25y_{l,\mathrm{STD},t-2}\ \le\ 46.
\end{align*}
\note{Take $y_{lsv}=0$ for $v\le0$ (empty initial lockers).}
\subhead{Worked example: Hopkins, day 2}
\[O_{\mathrm{H},2}=20+7+0.25(18)+0.70(20)=45.5\le46.\]
\note{Day-2 arrivals use 27 slots; day-1 carryover uses 18.5.}}

\panel{428}{570}{5}{Outputs and independent subproblems}
\block{432}{596}{381}{139}{%
\begin{align*}
N&=\sum_{l,s,t}d_{lst}=909,&
A&=\sum_{l,s,t}y^*_{lst}=803,\\
R&=N-A=106,&\eta&=\frac{A}{N}\approx88.3\%,\\
U_{lt}&=C_l-O_{lt}\ge0.&
\end{align*}
\note{$R$: rejected requests; $\eta$: acceptance rate; $U_{lt}$: unused expected capacity. A star denotes an optimal solution.}
\[Z^*(C,q)=\sum_{l\in\mathcal L}Z_l^*(C_l,q).\]
\note{The feasible set separates by locker. The full model is built in Pyomo and solved to optimality with HiGHS.}}

\panel{428}{982}{7}{Interpretation and assumptions}
\block{432}{1007}{381}{95}{%
\[C'\ge C\ \Longrightarrow\ Z^*(C',q)\ge Z^*(C,q).\]
\note{Increasing capacity enlarges the feasible set. Slower pickups increase expected occupancy and can reduce throughput.}
\textbf{Assumptions.} Empty start; one package per slot; no size classes; deliveries before pickups; common dwell probabilities; capacity enforced in expectation; no penalty after day 7.}
